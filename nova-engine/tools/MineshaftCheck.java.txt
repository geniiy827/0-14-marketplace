package com.nova.engine.world;
import java.util.*;
public class MT {
  public static void main(String[] a) throws Exception {
    int ss = Integer.parseInt(a[0]), cx = Integer.parseInt(a[1]), cz = Integer.parseInt(a[2]), pp = Integer.parseInt(a[3]);
    for (int i : new int[]{1, 3, 5, 7}) Blocks.solid[i] = true;
    Mineshaft.Start s = Mineshaft.start(new McRandom(ss), cx, cz, "1".equals(System.getenv("SURF")));
    final HashMap<String, int[]> W = new HashMap<>();
    Mineshaft.World w = new Mineshaft.World() {
      int[] w0(int x, int y, int z) { if (y <= 0) return new int[]{7,0}; if (y > 70) return new int[]{0,0}; if (Math.floorMod(x*7+z*13+y*3, 11) == 0) return new int[]{0,0}; return new int[]{1,0}; }
      public int get(int x, int y, int z) { int[] b = W.get(x+","+y+","+z); return (b != null ? b : w0(x,y,z))[0]; }
      public void set(int x, int y, int z, int id, int d) { W.put(x+","+y+","+z, new int[]{id, d}); }
      public int sky(int x, int y, int z) { return y >= 30 ? 15 : 0; }
    };
    McRandom r = new McRandom(pp);
    Mineshaft.Box big = new Mineshaft.Box(-100000, -100, -100000, 100000, 512, 100000);
    StringBuilder sb = new StringBuilder("{\"pieces\":[");
    boolean first = true;
    int NP = Integer.parseInt(System.getenv().getOrDefault("NP", "100000")), np = 0;
    for (Mineshaft.Piece p : s.pieces) {
      System.err.println("MTI " + np + " " + r.debugIndex());
      Mineshaft.DEBUG = np == NP - 1;
      if (np++ < NP) Mineshaft.postProcess(w, p, big, r);
      if (!first) sb.append(","); first = false;
      sb.append("{\"t\":").append(p.type).append(",\"dir\":").append(p.dir).append(",\"g\":").append(p.depth).append(",\"bb\":[").append(p.bb.x0).append(",").append(p.bb.y0).append(",").append(p.bb.z0).append(",").append(p.bb.x1).append(",").append(p.bb.y1).append(",").append(p.bb.z1).append("],\"x\":[").append(p.rails?1:0).append(",").append(p.sections).append(",").append(p.spiders?1:0).append(",").append(p.twoFloors?1:0).append("]}");
    }
    sb.append("],\"blocks\":{");
    first = true;
    for (Map.Entry<String,int[]> e : W.entrySet()) { if (!first) sb.append(","); first = false; sb.append("\"").append(e.getKey()).append("\":[").append(e.getValue()[0]).append(",").append(e.getValue()[1]).append("]"); }
    sb.append("}}");
    System.out.println(sb);
  }
}
